# q2 (Quarto v2) support Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let qvm install, use, list, and remove Quarto v2 ("q2", from `quarto-dev/q2`) alongside classic Quarto ("q1"), selected with a `--q2` flag, with q2 on PATH as `q2` and q1 unchanged as `quarto`.

**Architecture:** Introduce an `internal/edition` package whose `Edition` value captures every q1/q2 difference (repo, asset naming, archive layout, binary name, versions dir, checksum policy). The `config`, `gh`, `pipeline`, and `cmd` packages take an `Edition` parameter instead of hardcoding q1. Default edition is q1, preserving today's behavior exactly. q2 downloads are verified against their published `.sha256` checksums.

**Tech Stack:** Go, Cobra/Viper (CLI), go-github (releases), testify (tests), Just (build/test runner).

---

## Design reference

Spec: `docs/superpowers/specs/2026-06-16-q2-support-design.md`

Key facts confirmed against live q2 releases (`quarto-dev/q2` v0.1.1):

- q2 assets: `q2-{ver}-darwin_amd64.tar.gz`, `-darwin_arm64.tar.gz`, `-linux_amd64.tar.gz`, `-linux_arm64.tar.gz`, `-windows_amd64.zip`. No rhel7, no universal mac, no windows/arm64.
- q2 archive contains a single binary at the root: `q2` (tar.gz) / `q2.exe` (zip). No `bin/` tree.
- The existing `internal/unarchive` only strips a top dir when the entry name starts with `quarto-`, so q2's bare binary extracts directly into the version dir with mode preserved. **No unarchive changes needed.**
- q2 `.sha256` files have standard format: `<hexdigest>  <filename>`.

```go
// Package edition captures everything that differs between Quarto editions:
// where to fetch them, how their release assets are named, how their archives
// are laid out, and where/under what name each lives once installed.
package edition

import (
	"fmt"
	"path/filepath"
	"runtime"
)

// Kind identifies a Quarto edition. The zero value is Q1 so code paths that
// never set an edition keep classic-Quarto behavior.
type Kind int

const (
	Q1 Kind = iota // classic quarto (quarto-dev/quarto-cli)
	Q2             // quarto v2 (quarto-dev/q2)
)

// Platform is the resolved target we are installing for.
type Platform struct {
	OS   string // "darwin" | "linux" | "windows" | "rhel7"
	Arch string // "amd64" | "arm64"
}

// Edition captures where to fetch an edition, how its assets are named, and
// where/under what name it lives once installed.
type Edition struct {
	Kind            Kind
	Owner           string // GitHub owner, e.g. "quarto-dev"
	Repo            string // GitHub repo, e.g. "quarto-cli" or "q2"
	VersionsDirName string // dir under qvm data home, "versions" | "versions-q2"
	VerifyChecksum  bool   // verify the ".sha256" asset before extraction
}

// For returns the Edition for a Kind.
func For(k Kind) Edition {
	if k == Q2 {
		return Edition{
			Kind:            Q2,
			Owner:           "quarto-dev",
			Repo:            "q2",
			VersionsDirName: "versions-q2",
			VerifyChecksum:  true,
		}
	}
	return Edition{
		Kind:            Q1,
		Owner:           "quarto-dev",
		Repo:            "quarto-cli",
		VersionsDirName: "versions",
		VerifyChecksum:  false,
	}
}

// FromFlag maps the --q2 boolean to an edition.
func FromFlag(q2 bool) Edition {
	if q2 {
		return For(Q2)
	}
	return For(Q1)
}

// ExeName is the binary's name on PATH (the active symlink), OS-aware.
func (e Edition) ExeName() string {
	if e.Kind == Q2 {
		if runtime.GOOS == "windows" {
			return "q2.exe"
		}
		return "q2"
	}
	if runtime.GOOS == "windows" {
		return "quarto.cmd"
	}
	return "quarto"
}

// ExeRelPath is the executable's path inside an extracted version dir, OS-aware.
// q1 ships a directory tree (bin/quarto); q2 ships a single binary at the root.
func (e Edition) ExeRelPath() string {
	if e.Kind == Q2 {
		return e.ExeName() // "q2" / "q2.exe"
	}
	return filepath.Join("bin", e.ExeName()) // "bin/quarto" / "bin/quarto.cmd"
}

// AssetSuffix resolves the release-asset filename suffix for a platform.
// Callers match release assets by this suffix (strings.HasSuffix).
func (e Edition) AssetSuffix(p Platform) (string, error) {
	if e.Kind == Q2 {
		return q2AssetSuffix(p)
	}
	return q1AssetSuffix(p)
}

func q1AssetSuffix(p Platform) (string, error) {
	switch p.OS {
	case "darwin":
		return "macos.tar.gz", nil // universal build
	case "linux":
		if p.Arch == "arm64" {
			return "linux-arm64.tar.gz", nil
		}
		return "linux-amd64.tar.gz", nil
	case "windows":
		return "win.zip", nil
	case "rhel7":
		return "linux-rhel7-amd64.tar.gz", nil
	}
	return "", fmt.Errorf("quarto has no build for %s/%s", p.OS, p.Arch)
}

func q2AssetSuffix(p Platform) (string, error) {
	switch p.OS {
	case "darwin", "linux":
		if p.Arch != "amd64" && p.Arch != "arm64" {
			return "", fmt.Errorf("q2 has no build for %s/%s", p.OS, p.Arch)
		}
		return fmt.Sprintf("%s_%s.tar.gz", p.OS, p.Arch), nil
	case "windows":
		if p.Arch != "amd64" {
			return "", fmt.Errorf("q2 has no build for %s/%s", p.OS, p.Arch)
		}
		return "windows_amd64.zip", nil
	}
	return "", fmt.Errorf("q2 has no build for %s/%s", p.OS, p.Arch)
}
```

```go
	path, err := filepath.EvalSymlinks(GetPathToActiveExe(e))
```

```go
// GetInstalledVersions returns a map of installed versions for an edition where
// the key is the version and the value is the path to its executable.
func GetInstalledVersions(e edition.Edition) (map[string]string, error) {
	iv := make(map[string]string)
	entries, err := os.ReadDir(GetPathToVersionsDir(e))
	if err != nil {
		return iv, err
	}
	for _, entry := range entries {
		if entry.IsDir() {
			exePath := filepath.Join(GetPathToVersionsDir(e), entry.Name(), e.ExeRelPath())
			if _, err := os.Stat(exePath); err == nil {
				iv[entry.Name()] = exePath
			} else {
				log.Warn("could not find expected executable for version: ", entry.Name())
			}
		}
	}
	return iv, nil
}
```

```go
type WriteCounter struct {
	Written  int
	Total    int
	Label    string
	nWrites  int
	progress bool
}

func (wc *WriteCounter) Write(p []byte) (int, error) {
	n := len(p)
	wc.Written += n
	wc.nWrites++
	if wc.progress && wc.nWrites%300 == 0 {
		log.Infof("downloading quarto version %s ... (%s/%s)", wc.Label,
			humanize.Bytes(uint64(wc.Written)),
			humanize.Bytes(uint64(wc.Total)))
	}
	if wc.Written == wc.Total {
		log.Infof("completed downloading quarto version %s", wc.Label)
	}
	return n, nil
}

// DownloadReleaseAsset downloads the release asset for an edition+platform to a
// temp file and returns the path to the written file. When the edition requires
// it, the download is verified against its ".sha256" asset before returning.
func DownloadReleaseAsset(client *github.Client, e edition.Edition, tag string, p edition.Platform, progress bool) (string, error) {
	suffix, err := e.AssetSuffix(p)
	if err != nil {
		return "", err
	}
	release, err := GetRelease(client, e.Owner, e.Repo, tag)
	if err != nil {
		return "", fmt.Errorf("failed to get release %s: %w", tag, err)
	}
	asset := findAssetBySuffix(release.Assets, suffix)
	if asset == nil {
		return "", fmt.Errorf("no release asset found for %s/%s with suffix '%s'. This version may not support your platform/architecture. Available assets: %s",
			p.OS, p.Arch, suffix, listAvailableAssets(release.Assets))
	}
	log.Tracef("fetching information to download release asset from %s\n", asset.GetBrowserDownloadURL())
	start := time.Now()
	rc, _, err := client.Repositories.DownloadReleaseAsset(context.Background(), e.Owner, e.Repo, asset.GetID(), http.DefaultClient)
	log.Tracef("done fetching release asset information in %s\n", time.Since(start))
	if err != nil {
		return "", err
	}
	defer rc.Close()
	tmpFile, err := os.CreateTemp("", fmt.Sprintf("*-%s", asset.GetName()))
	if err != nil {
		return "", err
	}
	defer tmpFile.Close()
	log.Tracef("starting to copy release asset to %s\n", tmpFile.Name())
	start = time.Now()
	counter := &WriteCounter{Total: asset.GetSize(), Label: tag, progress: progress}
	wb, err := io.Copy(tmpFile, io.TeeReader(rc, counter))
	log.Tracef("done copying release asset in %s\n", time.Since(start))
	if err != nil {
		return tmpFile.Name(), err
	}
	log.Debugf("downloaded %d bytes from release asset at url %s\n", wb, asset.GetBrowserDownloadURL())

	if e.VerifyChecksum {
		if err := verifyAssetChecksum(client, e, release.Assets, asset.GetName(), tmpFile.Name()); err != nil {
			return "", err
		}
		log.Infof("verified sha256 checksum for %s", asset.GetName())
	}
	return tmpFile.Name(), nil
}

func findAssetBySuffix(assets []*github.ReleaseAsset, suffix string) *github.ReleaseAsset {
	if suffix == "" {
		return nil
	}
	for _, asset := range assets {
		if strings.HasSuffix(asset.GetName(), suffix) {
			return asset
		}
	}
	return nil
}

func listAvailableAssets(assets []*github.ReleaseAsset) string {
	var names []string
	for _, asset := range assets {
		name := asset.GetName()
		if strings.HasSuffix(name, ".tar.gz") || strings.HasSuffix(name, ".zip") {
			names = append(names, name)
		}
	}
	if len(names) == 0 {
		return "none"
	}
	return strings.Join(names, ", ")
}

// verifyAssetChecksum downloads the "<assetName>.sha256" sibling asset and
// verifies the downloaded file against it.
func verifyAssetChecksum(client *github.Client, e edition.Edition, assets []*github.ReleaseAsset, assetName, filePath string) error {
	checksumName := assetName + ".sha256"
	var checksumAsset *github.ReleaseAsset
	for _, a := range assets {
		if a.GetName() == checksumName {
			checksumAsset = a
			break
		}
	}
	if checksumAsset == nil {
		return fmt.Errorf("checksum asset %s not found; refusing to install unverified download", checksumName)
	}
	rc, _, err := client.Repositories.DownloadReleaseAsset(context.Background(), e.Owner, e.Repo, checksumAsset.GetID(), http.DefaultClient)
	if err != nil {
		return err
	}
	defer rc.Close()
	data, err := io.ReadAll(rc)
	if err != nil {
		return err
	}
	return verifyFileChecksum(filePath, string(data))
}

// verifyFileChecksum compares the SHA256 of filePath against the first
// whitespace-separated field of a standard "<hexdigest>  <name>" checksum line.
func verifyFileChecksum(filePath, checksumContent string) error {
	fields := strings.Fields(checksumContent)
	if len(fields) == 0 {
		return fmt.Errorf("malformed checksum content for %s", filePath)
	}
	want := fields[0]
	got, err := sha256File(filePath)
	if err != nil {
		return err
	}
	if !strings.EqualFold(got, want) {
		return fmt.Errorf("checksum mismatch for %s: expected %s, got %s", filePath, want, got)
	}
	return nil
}

func sha256File(path string) (string, error) {
	f, err := os.Open(path)
	if err != nil {
		return "", err
	}
	defer f.Close()
	h := sha256.New()
	if _, err := io.Copy(h, f); err != nil {
		return "", err
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}
```

```go
// DownloadReleaseVersion downloads and extracts the appropriate asset for an
// edition and platform into the edition's versions dir, returning the install
// directory.
func DownloadReleaseVersion(e edition.Edition, release string, p edition.Platform, progress bool) (string, error) {
	client := gh.NewClient(os.Getenv("GITHUB_PAT"))
	dlPath, err := gh.DownloadReleaseAsset(client, e, release, p, progress)

	}

	if err != nil {
		return "", err
	}
	err = unarchive.Unarchive(dlFile, dirPath)
	if err != nil {
		return "", err
	}
	return dirPath, err
}
```
